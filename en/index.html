agente_sql.py 
# question in natural language
$ python agente_sql.py
"Which customers bought the most in 2026?"

# query generated by the agent
SELECT c.nome, SUM(p.valor) AS total
FROM clientes c
JOIN pedidos p ON p.cliente_id = c.id
WHERE p.ano = 2026
GROUP BY c.nome
ORDER BY total DESC
LIMIT 5;
