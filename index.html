agente_sql.py 
# pergunta em linguagem natural
$ python agente_sql.py
"Quais clientes mais compraram em 2026?"

# consulta gerada pelo agente
SELECT c.nome, SUM(p.valor) AS total
FROM clientes c
JOIN pedidos p ON p.cliente_id = c.id
WHERE p.ano = 2026
GROUP BY c.nome
ORDER BY total DESC
LIMIT 5;
